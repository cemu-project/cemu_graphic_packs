int clampFI32(int v)
{
if( v == 0x7FFFFFFF )
	return floatBitsToInt(1.0);
else if( v == 0xFFFFFFFF )
	return floatBitsToInt(0.0);
return floatBitsToInt(clamp(intBitsToFloat(v), 0.0, 1.0));
}
float mul_nonIEEE(float a, float b){ if( a == 0.0 || b == 0.0 ) return 0.0; return a*b; }

if( activeMaskStackC[1] == true ) {
// 0

R123i.y = floatBitsToInt(mul_nonIEEE(intBitsToFloat(R2i.x),intBitsToFloat(uf_remappedPS[0].x)) + intBitsToFloat(R1i.x));
R2i.z = floatBitsToInt(1.0);

R0i.z = 0;
// 1

R0i.xyzw = floatBitsToInt(texture(textureUnitPS2, vec2(intBitsToFloat(R0i.x),intBitsToFloat(R0i.y))).xyzw);
// --- Exposure and Foliage Fix: replacement for the game's auto exposure ----------------------
// The game adapts its exposure on the CPU from a 64x64 luminance image. Cemu does not copy that
// render target back into guest memory, so the adaptation never happens and daylight scenes are
// far too bright. Here the mean brightness is measured from the blurred bloom texture (PS2), which
// already contains the game exposure. Scene and bloom are scaled down by the same factor.
{
	// 32 samples of the bloom texture. Single outliers (sparks, lamps) are limited to a multiple
	// (preset variable outlierCap) of the mean so that they do not distort the measurement.
	float expTap[32];
	float expSum = 0.0;
	for (int expI = 0; expI < 32; expI++)
	{
		vec2 expUV = vec2((float(expI % 8) + 0.5) / 8.0, (float(expI / 8) + 0.5) / 4.0);
		float expL = dot(max(textureLod(textureUnitPS2, expUV, 0.0).xyz, vec3(0.0)), vec3(0.299, 0.587, 0.114));
		if (isnan(expL) || isinf(expL)) expL = 32.0;
		expTap[expI] = expL;
		expSum += expL;
	}
	float expLimit = float($outlierCap) * expSum / 32.0;
	float expSum2 = 0.0;
	for (int expI = 0; expI < 32; expI++) expSum2 += min(expTap[expI], expLimit);
	float expMean = max(expSum2 / 32.0, 0.001);
	// Response curve: damped (factor = target * mean^-response), but the resulting image brightness
	// never exceeds target * ceiling. Very bright scenes (facing the sun, tunnel exits) are fully compensated.
	float expFactor = min(float($target) * pow(expMean, -float($response)), float($target) * float($ceiling) / expMean);
	expFactor = clamp(expFactor, float($minFactor), float($maxFactor));
	// Stylised cutscenes (for example the negative look of some race intros) are graded with a colour
	// table that only works with the exposure the game sets for them. With the correction they turn into
	// a black image with a green sky. They use a neutral grey bloom colour together with an exposure far
	// below the values used while driving (5.8 to 6.6 instead of 9.9 to 14.7) and are left untouched.
	vec3 expTint = vec3(intBitsToFloat(uf_remappedPS[1].x), intBitsToFloat(uf_remappedPS[1].y), intBitsToFloat(uf_remappedPS[1].z));
	if (max(abs(expTint.x - expTint.y), abs(expTint.y - expTint.z)) < 0.002 && intBitsToFloat(uf_remappedPS[2].w) < 8.5) expFactor = 1.0;
	exposureK = expFactor;
	R0i.xyz = floatBitsToInt(intBitsToFloat(R0i.xyz) * exposureK);
}
// --- end of exposure replacement -----------------------------------------------------------------

// --- Exposure and Foliage Fix: optional highlight compression -------------------------------
// The game's tone curve is y = 1 - 2^(-x). x is recovered here and compressed above a knee.
// Midtones stay unchanged, very bright parts such as the sky become darker and keep more colour.
// With highlights = 0 this block does nothing.
if (float($highlights) > 0.0)
{
	vec3 expX = -log2(max(1.0 - clamp(intBitsToFloat(R1i.xyz), vec3(0.0), vec3(0.9999)), vec3(0.0001)));
	float expLum = dot(expX, vec3(0.299, 0.587, 0.114));
	float expKnee = float($knee);
	if (expLum > expKnee)
	{
		float expNew = expKnee + (expLum - expKnee) / (1.0 + float($highlights) * (expLum - expKnee));
		expX *= expNew / expLum;
		R1i.xyz = floatBitsToInt(1.0 - exp2(-expX));
	}
}
