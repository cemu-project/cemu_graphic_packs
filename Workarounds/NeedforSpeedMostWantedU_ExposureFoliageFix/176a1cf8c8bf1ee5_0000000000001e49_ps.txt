int clampFI32(int v)
{
if( v == 0x7FFFFFFF )
	return floatBitsToInt(1.0);
else if( v == 0xFFFFFFFF )
	return floatBitsToInt(0.0);
return floatBitsToInt(clamp(intBitsToFloat(v), 0.0, 1.0));
}
float mul_nonIEEE(float a, float b){ if( a == 0.0 || b == 0.0 ) return 0.0; return a*b; }
void main()
{
vec4 R0f = vec4(0.0);
vec4 R1f = vec4(0.0);
vec4 R2f = vec4(0.0);
vec4 R3f = vec4(0.0);
vec4 R4f = vec4(0.0);
vec4 R123f = vec4(0.0);
vec4 R126f = vec4(0.0);
vec4 R127f = vec4(0.0);
float backupReg0f, backupReg1f, backupReg2f, backupReg3f, backupReg4f;
float PV0fx = 0.0, PV0fy = 0.0, PV0fz = 0.0, PV0fw = 0.0, PV1fx = 0.0, PV1fy = 0.0, PV1fz = 0.0, PV1fw = 0.0;
float PS0f = 0.0, PS1f = 0.0;
vec4 tempf = vec4(0.0);
float tempResultf;
int tempResulti;
ivec4 ARi = ivec4(0);
bool predResult = true;
R0f = passParameterSem133;
R1f = passParameterSem134;
R2f = passParameterSem135;
R3f = passParameterSem136;
R2f.xyz = (texture(textureUnitPS2, vec2(R2f.x,R2f.y)).xyz);
// --- Exposure and Foliage Fix: match the lens dirt overlay to the corrected exposure --------
// Both the bloom part (PS2) and the sun part (R1f) contain the excessive game exposure. The sun
// part arrives tone mapped from the vertex shader: y = 1 - 2^(-x*E). With the factor k this
// becomes 1 - (1 - y)^k, computed in linear space.
{
	// 32 samples of the bloom texture. Single outliers (sparks, lamps) are limited to a multiple
	// (preset variable outlierCap) of the mean so that they do not distort the measurement.
	float expTap[32];
	float expSum = 0.0;
	for (int expI = 0; expI < 32; expI++)
	{
		vec2 expUV = vec2((float(expI % 8) + 0.5) / 8.0, (float(expI / 8) + 0.5) / 4.0);
		float expL = dot(max(textureLod(textureUnitPS2, expUV, 0.0).xyz, vec3(0.0)), vec3(0.299, 0.587, 0.114));
		if (isnan(expL) || isinf(expL)) expL = 32.0;
		expTap[expI] = expL;
		expSum += expL;
	}
	float expLimit = float($outlierCap) * expSum / 32.0;
	float expSum2 = 0.0;
	for (int expI = 0; expI < 32; expI++) expSum2 += min(expTap[expI], expLimit);
	float expMean = max(expSum2 / 32.0, 0.001);
	// Response curve: damped (factor = target * mean^-response), but the resulting image brightness
	// never exceeds target * ceiling. Very bright scenes (facing the sun, tunnel exits) are fully compensated.
	float expFactor = min(float($target) * pow(expMean, -float($response)), float($target) * float($ceiling) / expMean);
	expFactor = clamp(expFactor, float($minFactor), float($maxFactor));
	R2f.xyz *= expFactor;
	vec3 expSun = pow(clamp(R1f.xyz, vec3(0.0), vec3(0.999)), vec3(2.2));
	expSun = 1.0 - pow(1.0 - expSun, vec3(expFactor));
	R1f.xyz = pow(expSun, vec3(1.0 / 2.2));
}
// --- end of exposure replacement -----------------------------------------------------------------
R4f.xyzw = (texture(textureUnitPS0, vec2(R0f.x,R0f.y)).xyzw);
// 0
R0f.x = intBitsToFloat(uf_remappedPS[0].x);
R0f.y = intBitsToFloat(uf_remappedPS[0].y);
PV0fz = mul_nonIEEE(R2f.x, intBitsToFloat(uf_remappedPS[1].x));
PV0fw = mul_nonIEEE(R2f.y, intBitsToFloat(uf_remappedPS[1].y));
// 1
backupReg0f = R2f.z;
R2f.y = PV0fw * intBitsToFloat(0x3ecccccd);
R2f.z = PV0fz * intBitsToFloat(0x3ecccccd);
R0f.w = mul_nonIEEE(backupReg0f, intBitsToFloat(uf_remappedPS[1].z));
R0f.xyz = (texture(textureUnitPS1, vec2(R0f.x,R0f.y)).xxx);
// 0
R127f.x = mul_nonIEEE(R1f.x,R0f.x) + R2f.z;
R127f.y = mul_nonIEEE(R1f.y,R0f.y) + R2f.y;
PV0fw = R0f.w * intBitsToFloat(0x3ecccccd);
// 1
R126f.x = mul_nonIEEE(R127f.y, R4f.y);
R123f.z = mul_nonIEEE(R1f.z,R0f.z) + PV0fw;
R127f.w = mul_nonIEEE(R127f.x, R4f.x);
// 2
tempf.x = dot(vec4(R127f.x,R127f.y,R123f.z,intBitsToFloat(0x80000000)),vec4(intBitsToFloat(0x3e991687),intBitsToFloat(0x3f1645a2),intBitsToFloat(0x3de978d5),0.0));
PV0fx = tempf.x;
PV0fy = tempf.x;
PV0fz = tempf.x;
PV0fw = tempf.x;
PS0f = mul_nonIEEE(R123f.z, R4f.z);
// 3
PV1fx = mul_nonIEEE(PV0fx, R4f.w);
R4f.y = mul_nonIEEE(R3f.y, R126f.x);
R4f.z = mul_nonIEEE(R3f.z, PS0f);
R4f.x = mul_nonIEEE(R3f.x, R127f.w);
// 4
R4f.w = mul_nonIEEE(R3f.w, PV1fx);
// export
passPixelColor0 = vec4(R4f.x, R4f.y, R4f.z, R4f.w);
}
